Using them

All of these work in any Palladium power file (data/<namespace>/palladium/powers/*.json) like Palladium's own types. Properties left out use the defaults listed here. A power that gives a custom stone portals in a fire style might look like this:

{
  "name": "Chaos Stone",
  "icon": "example:chaos_stone",
  "gui_display_type": "tree",
  "abilities": {
    "sling_ring": {
      "type": "infinity:portal",
      "title": "Sling Ring",
      "style": "fire",
      "size": 1.5,
      "conditions": {
        "enabling": [ { "type": "palladium:action", "key_type": "key_bind", "cooldown": 40 } ]
      }
    },
    "blast": {
      "type": "infinity:energy_blast",
      "title": "Chaos Blast",
      "damage": 8.0,
      "color": "#E0115F",
      "scale_with_power_level": false,
      "conditions": {
        "enabling": [ { "type": "palladium:action", "key_type": "key_bind", "cooldown": 10 } ],
        "unlocking": [ { "type": "infinity:gauntlet_slot", "item": { "item": "example:chaos_stone" } } ]
      }
    }
  }
}

Palladium's generated documentation (mods/documentation/) lists the same types with the exact property formats for your installed version.
